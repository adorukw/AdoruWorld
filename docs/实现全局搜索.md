2. CRUD（核心搜索逻辑）
关键词搜索的本质是：分割关键词 → 对每个关键词构建 or_ 条件 → 关键词之间用 and_ → 关联项通过 JOIN 搜索 → distinct 去重。

# crud.py  — 伪代码/思路
from sqlalchemy import or_, and_, select, func
from sqlalchemy.orm import selectinload
from sqlalchemy.ext.asyncio import AsyncSession

from app.modules.post.model import Post
from app.modules.post_category.model import PostCategory
from app.modules.post_tag.model import PostTag
from app.modules.dex.model import Dex
from app.modules.dex_genre.model import DexGenre
from app.modules.media.model import Media
from app.modules.media_tag.model import MediaTag


def _split_keywords(q: str) -> list[str]:
    """将查询字符串拆成多个关键词（空格/逗号分隔）"""
    import re
    return [kw.strip() for kw in re.split(r'[\s,，]+', q) if kw.strip()]


async def search_all(
    db: AsyncSession,
    q: str,
    entity_type: str | None = None,   # 可选筛 "post"/"dex"/"media"
    skip: int = 0,
    limit: int = 20,
) -> tuple[list[dict], int]:
    """
    核心搜索方法：对三个实体各自执行搜索，然后合并、排序、分页。
    返回 (items, total_count)
    """
    keywords = _split_keywords(q)
    if not keywords:
        return [], 0

    results = []  # 用来装 (created_at, type, data) 三元组方便统一排序

    # ===== 搜索 Post =====
    if entity_type in (None, "post"):
        post_conditions = []
        for kw in keywords:
            post_conditions.append(
                or_(
                    Post.title.ilike(f"%{kw}%"),
                    Post.description.ilike(f"%{kw}%"),
                    Post.content.ilike(f"%{kw}%"),
                    PostCategory.name.ilike(f"%{kw}%"),
                    PostTag.name.ilike(f"%{kw}%"),
                )
            )

        stmt = (
            select(Post)
            .options(selectinload(Post.category), selectinload(Post.tags))
            .outerjoin(Post.tags)        # 为了跨标签名搜索
            .outerjoin(Post.category)    # 为了跨分类名搜索
            .where(and_(*post_conditions))
            .distinct()
            # ⚠️ distinct 必须在 outerjoin 后，且 order_by 的列必须也在 SELECT 中
            .order_by(Post.created_at.desc())
        )
        res = await db.execute(stmt)
        posts = res.scalars().all()
        for p in posts:
            results.append((p.created_at, "post", p))

    # ===== 搜索 Dex =====
    if entity_type in (None, "dex"):
        dex_conditions = []
        for kw in keywords:
            dex_conditions.append(
                or_(
                    Dex.title.ilike(f"%{kw}%"),
                    Dex.original_title.ilike(f"%{kw}%"),
                    Dex.summary.ilike(f"%{kw}%"),
                    Dex.comment.ilike(f"%{kw}%"),
                    Dex.creator.ilike(f"%{kw}%"),
                    DexGenre.name.ilike(f"%{kw}%"),
                )
            )

        stmt = (
            select(Dex)
            .options(selectinload(Dex.genres))
            .outerjoin(Dex.genres)
            .where(and_(*dex_conditions))
            .distinct()
        )
        res = await db.execute(stmt)
        dexs = res.scalars().all()
        # Dex 没有 created_at，用 title 排序作为 fallback
        for d in dexs:
            results.append((None, "dex", d))

    # ===== 搜索 Media =====
    if entity_type in (None, "media"):
        media_conditions = []
        for kw in keywords:
            media_conditions.append(
                or_(
                    Media.title.ilike(f"%{kw}%"),
                    Media.file_path.ilike(f"%{kw}%"),
                    MediaTag.name.ilike(f"%{kw}%"),
                )
            )

        stmt = (
            select(Media)
            .options(selectinload(Media.tags))
            .outerjoin(Media.tags)
            .where(and_(*media_conditions))
            .distinct()
            .order_by(Media.uploaded_at.desc())
        )
        res = await db.execute(stmt)
        medias = res.scalars().all()
        for m in medias:
            results.append((m.uploaded_at, "media", m))

    # ===== 统一排序 =====
    results.sort(key=lambda x: (x[0] is None, x[0] or ""), reverse=False)

    total = len(results)
    page = results[skip: skip + limit]

    # ===== 转为通用响应格式 =====
    items = [_to_search_item(r) for r in page]
    return items, total


def _to_search_item(item_tuple) -> SearchResultItem:
    """将 (sort_key, type, entity) 转换为 SearchResultItem"""
    sort_key, etype, entity = item_tuple
    cover = getattr(entity, "cover_image", None) or getattr(entity, "file_path", None)

    return SearchResultItem(
        id=entity.id,
        type=etype,
        title=entity.title,
        slug=entity.slug,
        description=getattr(entity, "description", None)
                  or getattr(entity, "summary", None),
        cover_image=cover,
        created_at=str(sort_key) if sort_key else None,
        entity_data=_entity_to_dict(etype, entity),
    )


def _entity_to_dict(etype: str, entity) -> dict:
    """基于已知模式的转换，避免暴露敏感字段"""
    if etype == "post":
        return {
            "category": entity.category.name if entity.category else None,
            "tags": [t.name for t in entity.tags],
            "published": entity.published,
        }
    elif etype == "dex":
        return {
            "original_title": entity.original_title,
            "category": entity.category,
            "genres": [g.name for g in entity.genres],
            "rating": entity.rating,
        }
    elif etype == "media":
        return {
            "media_type": entity.media_type,
            "file_path": entity.file_path,
            "tags": [t.name for t in entity.tags],
        }
    return {}

# router.py
from fastapi import APIRouter, Depends, Query
from sqlalchemy.ext.asyncio import AsyncSession
from app.core.database import get_db
from .schema import SearchResponse
from . import crud

router = APIRouter(prefix="/search", tags=["search"])


@router.get("", response_model=SearchResponse)
async def global_search(
    q: str = Query(..., min_length=1, description="搜索关键词"),
    type: str | None = Query(None, regex="^(post|dex|media)$"),
    skip: int = Query(0, ge=0),
    limit: int = Query(20, ge=1, le=100),
    db: AsyncSession = Depends(get_db),
):
    items, total = await crud.search_all(db, q, type, skip, limit)
    return SearchResponse(items=items, total=total, skip=skip, limit=limit)

⚠️ 关键细节和陷阱
outerjoin + distinct 的必要性
一个 Post 有多个 Tag，如果你做 .join(Post.tags)，一条 Post 会被复制成 N 行（每匹配一个 Tag 就一行）。这时候必须：

用 outerjoin 而不是 join（否则没有 Tag 的 Post 不会被搜到）
用 .distinct() 去重
selectinload 与 join 的兼容性
SQLAlchemy 的 selectinload 会在主查询之后单独发第二条 SQL 加载关联。它不影响 join/outerjoin 的筛选逻辑，所以可以用在一起。

多关键词的 AND 逻辑
如果用户搜索 "RPG 奇幻"，你想找到同时包含这两个词的结果。上面的代码通过 and_(*conditions) 实现了这点——每个关键词的所有 or_ 字段必须至少有一个匹配。

SQLite 的 LIKE 和中文
SQLite 的 LIKE 只对 ASCII 字符不区分大小写。对中文（CJK）它是精确匹配。如果要更好的中文搜索体验，有两个进阶方案：

SQLite FTS5：为每个表创建 FTS5 虚拟表，用 MATCH 做全文检索——性能好，支持中文分词（但要装 ICU tokenizer 或使用简单 trie-based tokenizer）。
Python 端正则回退：如果数据量不大（<万条级别），LIKE %keyword% 完全够用。
meta_data (JSON) 搜索
Media 的 meta_data 是 JSON 列。在 SQLite 中它存成文本，可以 cast 成 String 后搜索：

复制
Media.meta_data.cast(String).ilike(f"%{kw}%")
但这样会搜索到 JSON 的 key 字面量（比如 "width"、"height"），可能不够精确。建议只搜索用户可见的字段，meta_data 跳过搜索，除非你有具体需求。

总结建议
方案	优点	缺点
独立 search 模块（推荐）	一个端点搜所有，前端一个输入框就行，统一响应格式	多一个模块
每个路由加 keywords	改动最小，跟现有 API 结构一致	前端要调三个 API 再合并
先做独立 search，再考虑要不要给每个路由也补 keywords	最灵活	稍微多一点代码
个人建议先做独立 search 模块，因为你的前端就想要一个输入框、一个结果列表，走一个 API 是最自然的。响应格式统一后，前端渲染也简单——根据 type 字段显示不同的卡片模板就行。
