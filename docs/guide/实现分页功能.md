改动 `get_dex` 函数：

```python
async def get_dex(
        db: AsyncSession,
        category: str | None = None,
        status: str | None = None,
        skip: int = 0,
        limit: int = 20
) -> tuple[list[Dex], int]:          # ← 改返回值类型

    # 改为用两个查询（也可以 count 子查询，但简单场景双查即可）
    count_stmt = select(func.count()).select_from(Dex)
    list_stmt = select(Dex)

    if category:
        count_stmt = count_stmt.where(Dex.category == category)
        list_stmt = list_stmt.where(Dex.category == category)
    if status:
        count_stmt = count_stmt.where(Dex.status == status)
        list_stmt = list_stmt.where(Dex.status == status)

    # 查总数
    total_res = await db.execute(count_stmt)
    total = total_res.scalar_one()

    # 查当前页数据
    list_stmt = list_stmt.order_by(Dex.title).offset(skip).limit(limit)
    list_res = await db.execute(list_stmt)

    return list(list_res.scalars().all()), total   # ← 返回元组
```

### 5. `client/src/store/dex.ts` — 改 store，加分页状态

```typescript
import { defineStore } from "pinia";
import { ref, computed } from "vue";
import { api } from "@/api";
import type { DexResponse, DexStats, DexCreate, DexUpdate, DexListResponse } from "@/types";  // ← 加 DexListResponse

export const useDexStore = defineStore("dex", () => {
  // ── 核心数据 ──
  const dexs = ref<DexResponse[]>([]);
  const currentDex = ref<DexResponse | null>(null);
  const dexStats = ref<DexStats>({
    total: 0,
    byCategory: {},
    byStatus: {},
    averageRating: 0,
  });

  // ── 分页状态 ──
  const currentPage = ref(1);
  const pageSize = ref(20);
  const totalItems = ref(0);

  const totalPages = computed(() => Math.ceil(totalItems.value / pageSize.value));

  // ── 加载状态 ──
  const loading = ref(false);
  const error = ref<string | null>(null);

  // ── 当前筛选参数（给后端过滤用） ──
  const activeCategory = ref<string | undefined>();
  const activeStatus = ref<string | undefined>();

  // ── 获取列表（分页 + 筛选全走后端） ──
  const getDexs = async (params?: {
    category?: string;
    status?: string;
    skip?: number;
    limit?: number;
  }) => {
    loading.value = true;
    error.value = null;
    try {
      const res: DexListResponse = await api.dexs.list(params);
      dexs.value = res.items;
      totalItems.value = res.total;
    } catch (err: any) {
      error.value = err.message || "获取图鉴列表失败";
    } finally {
      loading.value = false;
    }
  };

  // ── 切换页码 ──
  const goToPage = (page: number) => {
    if (page < 1 || page > totalPages.value) return;
    currentPage.value = page;
    getDexs({
      category: activeCategory.value,
      status: activeStatus.value,
      skip: (page - 1) * pageSize.value,
      limit: pageSize.value,
    });
  };

  // ── 应用筛选（重置到第 1 页） ──
  const applyFilters = (category?: string, status?: string) => {
    activeCategory.value = category === 'all' ? undefined : category;
    activeStatus.value = status === 'all' ? undefined : status;
    currentPage.value = 1;
    getDexs({
      category: activeCategory.value,
      status: activeStatus.value,
      skip: 0,
      limit: pageSize.value,
    });
  };

  // ── 其余方法不变 ──
  const getDexStats = async () => {
    loading.value = true;
    error.value = null;
    try {
      dexStats.value = await api.dexs.stats();
    } catch (err: any) {
      error.value = err.message || "获取图鉴统计失败";
    } finally {
      loading.value = false;
    }
  };

  const getDexBySlug = async (slug: string) => { /* 不变 */ };
  const getDexById = async (id: string) => { /* 不变 */ };
  const createDex = async (data: DexCreate) => { /* 不变 */ };
  const updateDex = async (id: string, data: DexUpdate) => { /* 不变 */ };
  const deleteDex = async (id: string) => { /* 不变 */ };

  return {
    dexs,
    dexStats,
    loading,
    error,

    // 分页
    currentPage,
    pageSize,
    totalItems,
    totalPages,
    goToPage,
    applyFilters,

    getDexs,
    getDexBySlug,
    getDexById,
    createDex,
    updateDex,
    deleteDex,
    getDexStats,
  };
});
```

  const total = dexStore.totalPages
  const current = dexStore.currentPage
  const delta = 2 // 当前页前后各显示 2 页
  const range: number[] = []
  const start = Math.max(2, current - delta)
  const end = Math.min(total - 1, current + delta)
  range.push(1)
  if (start > 2) range.push(0) // 0 表示省略号
  for (let i = start; i <= end; i++) range.push(i)
  if (end < total - 1) range.push(0)
  if (total > 1) range.push(total)
  return range

|------|-------------|---------|

**关键瓶颈：** 后端 `GET /posts` 和 `GET /dexs` 都只返回数据数组，**不返回总条数**。前端无法知道总页数，也就做不了标准的页码导航。

## 🛠 完整方案（两步走）

### 第一步：后端改造——让 API 返回分页元信息

当前后端返回是裸 `list[PostResponse]`，缺少分页必要字段：
```python
# 现状
@router.get("", response_model=list[PostResponse])  # 只有数据，没有页码

# 需要返回类似
{
    "items": [...],
    "total": 42,
    "skip": 0,
    "limit": 10
}
```

**方案：新建一个通用分页 Schema**

在 `server/app/schemas/post.py` 里追加（或用泛型）：

```python
# 分页包裹结构
from pydantic.generics import GenericModel
from typing import TypeVar, Generic

T = TypeVar("T")

class PaginatedResponse(GenericModel, Generic[T]):
    items: list[T]
    total: int
    skip: int
    limit: int
```

然后在 `routers/posts.py` 中改 `list_posts`：

```python
@router.get("", response_model=PaginatedResponse[PostResponse])
async def list_posts(
    published: bool | None = None,
    featured: bool | None = None,
    category: str | None = None,
    tag: str | None = None,
    skip: int = Query(0, ge=0),
    limit: int = Query(10, ge=1, le=100),
    db: AsyncSession = Depends(get_db)
):
    items = await crud.get_posts(...)
    total = await crud.get_total_posts_count(db, published_only=published if published is not None else True)
    return PaginatedResponse(items=[PostResponse(**format_post(p)) for p in items], total=total, skip=skip, limit=limit)
```

对 `GET /dexs` 做同样改造。在 `crud/dexs.py` 加一个 `get_dex_count`。

### 第二步：前端改造——在需要分页的页面加 UI

#### A) 通用分页组件 `Pagination.vue`

新建 `client/src/components/ui/Pagination.vue`：

```vue
<script setup lang="ts">
const props = defineProps<{
  total: number
  skip: number
  limit: number
}>()

const emit = defineEmits<{
  (e: 'page-change', skip: number): void
}>()

const currentPage = computed(() => Math.floor(props.skip / props.limit) + 1)
const totalPages = computed(() => Math.ceil(props.total / props.limit))
const hasPrev = computed(() => currentPage.value > 1)
const hasNext = computed(() => currentPage.value < totalPages.value)
</script>

<template>
  <div v-if="totalPages > 1" class="flex items-center justify-center gap-2 mt-8">
    <button :disabled="!hasPrev"
      class="px-3 py-1 border-2 disabled:opacity-40"
      @click="emit('page-change', skip - limit)">‹ 上一页</button>

    <span class="px-3 py-1 text-sm">
      {{ currentPage }} / {{ totalPages }}
    </span>

    <button :disabled="!hasNext"
      class="px-3 py-1 border-2 disabled:opacity-40"
      @click="emit('page-change', skip + limit)">下一页 ›</button>
  </div>
</template>
```

#### E) **改造 Dex.vue** —— 从全量客户端筛选改为服务端分页

当前 Dex.vue 把 `dexEntries` 全量拉到内存，然后在 `filteredEntries` 里做 `filter`。数据量大了以后应该改为：

1. 用 `skip`/`limit` 调用 `getDexEntries`，传入筛选参数
2. 将 `filteredEntries` 改为从 store 的当前分页数据 + 本地筛选
3. 加 Pagination 组件

或者更简单的过渡方案：先保持当前全量加载 + 客户端筛选，但加上 Pagination 对 `filteredEntries` 做**前端切片**：

```vue
<script setup lang="ts">
const pageSize = 20
const currentPage = ref(1)

const paginatedEntries = computed(() => {
  const start = (currentPage.value - 1) * pageSize
  return filteredEntries.value.slice(start, start + pageSize)
})

const totalPages = computed(() =>
  Math.ceil(filteredEntries.value.length / pageSize)
)
</script>

<template>
  <!-- 列表 -->
  ...
  <!-- 分页 -->
  <Pagination
    v-if="totalPages > 1"
    :total="filteredEntries.length"
    :skip="(currentPage - 1) * pageSize"
    :limit="pageSize"
    @page-change="(skip) => currentPage = Math.floor(skip / pageSize) + 1"
  />
</template>
```

> 等日后 Dex 数据量大了再切回**服务端分页**，那时只需把筛选参数传给 `getDexEntries` 即可。
